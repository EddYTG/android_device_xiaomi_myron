# [EN] Recovery binary directories | [CN] Recovery 主程序目录
# [VI] Thư mục binary Recovery | [RU] Каталоги Recovery
dir.recovery = /system/bin
dir.vendor_recovery = /vendor/bin
dir.vendor_recovery = /vendor/bin/hw
dir.odm_recovery = /odm/bin
dir.odm_recovery = /odm/bin/hw

# ============================================================================
# [recovery] namespace
# [EN] For core system components, allows cross-partition library access
# [VI] Dành cho thành phần hệ thống core, cho phép truy cập cross-partition
# [RU] Для основных системных компонентов, разрешает межразделный доступ
# [CN] 用于系统核心组件，允许跨分区访问 vendor/odm 库（解决 FBE 解密依赖）
# ============================================================================
[recovery]
namespace.default.isolated = false

# Library search paths | Đường dẫn tìm kiếm thư viện | Пути поиска библиотек | 库搜索路径
namespace.default.search.paths = /system/${LIB}
namespace.default.search.paths += /vendor/${LIB}
namespace.default.search.paths += /vendor/${LIB}/hw
namespace.default.search.paths += /odm/${LIB}

# ASan/HWAsan debug configs (disabled in Recovery, kept for compatibility)
# Cấu hình debug ASan (tắt trong Recovery, giữ để tương thích)
# Конфигурация отладки ASan (отключено в Recovery, для совместимости)
# ASan/HWAsan 调试配置（Recovery 模式通常禁用，保留兼容）
namespace.default.asan.search.paths = /data/asan/system/${LIB}
namespace.default.asan.search.paths += /system/${LIB}
namespace.default.hwasan.search.paths = /system/${LIB}/hwasan
namespace.default.hwasan.search.paths += /system/${LIB}

# ============================================================================
# [vendor_recovery] namespace  
# [EN] For vendor HALs (KeyMint, Gatekeeper, Weaver service)
# [VI] Dành cho vendor HALs (KeyMint, Gatekeeper, Weaver)
# [RU] Для vendor HALs (KeyMint, Gatekeeper, Weaver)
# [CN] 用于 vendor 分区 HAL 服务（如 KeyMint、Gatekeeper、指纹 Weaver）
# ============================================================================
[vendor_recovery]
namespace.default.isolated = false

# Allows vendor bins to access odm extensions and system libs
# Cho phép vendor truy cập odm extensions và system libs
# Разрешает vendor доступ к odm расширениям и system библиотекам
# 允许 vendor 二进制文件访问 odm 扩展库和系统基础库
namespace.default.search.paths = /vendor/${LIB}
namespace.default.search.paths += /vendor/${LIB}/hw
namespace.default.search.paths += /odm/${LIB}
namespace.default.search.paths += /system/${LIB}

# ============================================================================
# [odm_recovery] namespace
# [EN] For Xiaomi ODM customizations (charging HAL, displayfeature, etc.)
# [VI] Dành cho tùy chỉnh ODM Xiaomi (charging HAL, displayfeature)
# [RU] Для Xiaomi ODM кастомизаций (charging HAL, displayfeature)
# [CN] 用于小米 ODM 定制组件（充电 HAL、显示特性、振动触觉等）
# ============================================================================
[odm_recovery]
namespace.default.isolated = false

# ODM libs prioritized, fallback to vendor/system
# Ưu tiên ODM libs, fallback về vendor/system
# Приоритет ODM libs, fallback к vendor/system
# ODM 库优先搜索，回退到 vendor/system
namespace.default.search.paths = /odm/${LIB}
namespace.default.search.paths += /vendor/${LIB}
namespace.default.search.paths += /vendor/${LIB}/hw
namespace.default.search.paths += /system/${LIB}
